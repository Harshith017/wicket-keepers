// Add these inside the existing   match /databases/{database}/documents { ... }
// block in the Firebase console (Firestore Database > Rules), keeping the rules
// that are already there for the trip app. Replace OWNER_EMAIL_HERE with the
// Google account that should be able to approve people and publish the app.

function trackerOwner() {
  return request.auth != null
    && request.auth.token.email_verified == true
    && request.auth.token.email == 'OWNER_EMAIL_HERE';
}
function trackerAllowed() {
  return request.auth != null
    && request.auth.token.email_verified == true
    && exists(/databases/$(database)/documents/trackerUsers/$(request.auth.token.email));
}

// People the owner has approved. A person can only see their own entry.
match /trackerUsers/{email} {
  allow get: if request.auth != null && request.auth.token.email == email;
  allow read, write: if trackerOwner();
}

// People asking for access. Anyone signed in can create, see or withdraw
// their own request; only the owner sees the list.
match /trackerRequests/{email} {
  allow get, delete: if request.auth != null && request.auth.token.email == email;
  allow create, update: if request.auth != null
    && request.auth.token.email_verified == true
    && request.auth.token.email == email
    && request.resource.data.keys().hasOnly(['name', 'photo', 'requestedAt'])
    && request.resource.data.name is string
    && request.resource.data.name.size() > 0
    && request.resource.data.name.size() <= 40;
  allow read, delete: if trackerOwner();
}

// The app itself. Only approved people (and the owner) can download it.
match /trackerApp/{doc} {
  allow get: if trackerAllowed() || trackerOwner();
  allow write: if trackerOwner();
}
